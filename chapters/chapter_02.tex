\chapter{The Curry--Howard--Lambek correspondence}

From this chapter onward, we will only consider cartesian closed categories, for which we will use the abbreviations CCC and CCCs.

\section{Natural number objects in CCCs}
The aim of this section is to provide cartesian closed categories with a notion of recursion,
which is fundamental if we want to interpret a simply $\lambda$-calculus with iterator in a CCC.
Even though this notion won't be necessary for the correspondence used to prove Lawvere's fixed-point theorem,
it is still necessary to prove the Curry--Howard--Lambek correspondence in its general version, and nevertheless
interesting in its own right. One notable application of this notion is, for example, the interpretation of all primitive recursive functions in a CCC with a natural number object, see \cite[Part~III, \S\S~1--2]{LambekScott1994}.

\begin{definition}[Natural number object]
    Given a cartesian closed category $\mathcal{C}$, a \emph{natural number object} (NNO) is an initial object
    \[ 1 \xrightarrow{0} N \xrightarrow{s} N \]
    in the category of diagrams of the form $1 \xrightarrow{a} A \xrightarrow{f} A$ in $\mathcal{C}$. An equivalent way to define it is
    by saying that, for all such diagrams, there exists a unique arrow $h : N \to A$ such that the following diagram commutes:

    \[\begin{tikzcd}
        1 & N & N \\
        1 & A & A
        \arrow["0", from=1-1, to=1-2]

        \arrow["s", from=1-2, to=1-3]

        \arrow["f", from=2-2, to=2-3]
    \end{tikzcd},\]
    i.e., $h \circ 0 = a$ and $h \circ s = f \circ h$.
\end{definition}

If we require only the existence of an arrow that makes the diagram above commute, we obtain the following.

\begin{definition}[Weak natural number object]
    Given a cartesian closed category $\mathcal{C}$, a \emph{weak natural number object} (WNNO) is an object $N$ together with two arrows $1 \xrightarrow{0} N$ and $N \xrightarrow{s} N$ in $\mathcal C$ such that,
    for all diagrams of the form $1 \xrightarrow{a} A \xrightarrow{f} A$ in $\mathcal{C}$, there exists an arrow $h : N \to A$ that makes the following diagram commute:
    \[\begin{tikzcd}
        1 & N & N \\
        1 & A & A
        \arrow["0", from=1-1, to=1-2]
        \arrow[shift left=0.20ex, -{}, from=1-1, to=2-1]
        \arrow[shift right=0.20ex, -{}, from=1-1, to=2-1]
        \arrow["s", from=1-2, to=1-3]
        \arrow["h"', dashed, from=1-2, to=2-2]
        \arrow["h"', dashed, from=1-3, to=2-3]
        \arrow["a", from=2-1, to=2-2]
        \arrow["f", from=2-2, to=2-3]
    \end{tikzcd}.\]
\end{definition}

\begin{example}[NNO in $\textsf{Set}$]
    The category of sets $\textsf{Set}$, whose terminal object is any singleton set, has a natural number object given by the set of natural numbers $\mathbb{N}$ together with the arrows $0 : 1 \to \mathbb{N}$ and $s : \mathbb{N} \to \mathbb{N}$
    defined by $0(*) = 0$ and $s(n) = n + 1$, respectively. The existence and uniqueness of the arrow $h : \mathbb{N} \to A$ is guaranteed by the principle of countable recursion:
    \[ h(n) := \begin{cases}
        h(0) = a \\
        h(s(n)) = f(h(n))
    \end{cases}. \]
\end{example}

\begin{example}[WNNO which is not an NNO]
    Let $\textsf{Pos}$ be the category of partially ordered sets with monotone maps. Its terminal object, up to isomorphism, is the one-element poset; its products are cartesian products equipped with the lexicographic order,
    and its exponential objects are sets of monotone maps equipped with the pointwise order:
    \[  f \leq_{B^A} g \quad\overset{\textnormal{def}}{\Longleftrightarrow}\quad f(a) \leq_B g(a) \quad\text{for every } a \in A. \]
    The usual evaluation map is monotone, idem the currying map and (CCC1) + (CCC2) are easily verified. Thus $\textsf{Pos}$ is cartesian closed.
    Let $\omega_1$ be the first uncountable ordinal and let $W := (\omega_1)_{\mathrm{disc}}$ be its underlying set equipped with the discrete order, rather than with the usual ordinal order; explicitly:
    \[ \alpha \leq_W \beta \quad\overset{\textnormal{def}}{\Longleftrightarrow}\quad \alpha = \beta. \]
    We choose the arrows $0 : 1 \to W : * \mapsto 0$ and $s : W \to W : \alpha \mapsto \alpha + 1$, which is monotone and well-defined since $\omega_1$ is a limit ordinal.\newline
    Let's consider an arbitrary diagram $1 \xrightarrow{a} A \xrightarrow{f} A$ in $\textsf{Pos}$. Then we can define $h$ by transfinite recursion as follows:
    \[ h(\alpha) := \begin{cases}
        h(0) = a(*) \\
        h(s(\alpha)) = f(h(\alpha)) \\
        h(\lambda) = a(*) & \text{for $\lambda$ nonzero limit ordinal}
    \end{cases}. \]
    One can easily verify that $h$ is monotone (a map from a discrete-ordered poset it's always monotone). By definition, we have:
    \[ h \circ 0 = a \qquad\text{and}\qquad h \circ s = f \circ h,
    \]
    therefore $(W,0,s)$ is a WNNO in $\textsf{Pos}$.\newline
    This WNNO is not an NNO. In fact, let $2 = \{0,1\}$ have the discrete order, let $a : 1 \to 2$ select $0$, and take $1_2 : 2 \to 2$. Both the constant map $h_1 : W \to 2$ with value $0$ and the map
    \[
        h_2(\alpha) :=
        \begin{cases}
            1 & \text{if } \alpha = \omega + n \text{ for some } n \in \mathbb{N}, \\
            0 & \text{otherwise}
        \end{cases}
    \]
    are monotone and satisfy $h_i \circ 0 = a$ and $h_i \circ s = 1_2 \circ h_i$.
\end{example}

\begin{example}[NNO in $\textsf{Pos}$]
    $(\NN_{\mathrm{disc}}, 0, s)$, where $\NN_{\mathrm{disc}}$ is the set of natural numbers equipped with the discrete order, is an NNO in $\textsf{Pos}$, thanks to the principle of countable recursion.
    The reason why this is unique, whereas it need not to be unique for $(\omega_1)_{\mathrm{disc}}$, is that the commutativity of the diagram fixes $h(0) = a(*)$ and
    recursively defines $h(s(n))$ for every successor of $0 \in N$. Therefore, if we take an ordinal containing a nonzero limit ordinal,
    we obtain an “orbit” under the successor function, determined by a different initial value of $h$; however, the diagram above gives us only two conditions.
\end{example}

\begin{notation}

    Using this notation we can write the following map
    \[ J_A : \Hom(1,A) \times \Hom(A,A) \to \Hom(N,A) \]
    which assigns to each pair $(a,f)$ an arrow $J_A(a,f)$ that makes the diagram

        \arrow[shift right=0.20ex, -{}, from=1-1, to=2-1]

        \arrow["{J_A(a,f)}"', dashed, from=1-3, to=2-3]
        \arrow["a", from=2-1, to=2-2]
        \arrow["f", from=2-2, to=2-3]
    \end{tikzcd}\]
    commute, i.e.,\ $J_A(a,f) \circ 0 = a$ and $J_A(a,f) \circ s = f \circ J_A(a,f)$. For each object $A$, given $(a,f) \in \Hom(1,A) \times \Hom(A,A)$, we know
    by the definition of WNNO that $\Hom(N,A)$ is nonempty, therefore we can pick an element and define $J_A(\cdot,\cdot)$ as a function.
    In order to define $J_A(\cdot,\cdot)$ as a function, we need to choose simultaneously an element for each pair $(a,f)$, i.e.,\ to choose a function in $\prod_{(a,f)} \Hom(N,A)$.
    If $\mathcal C$ is \emph{locally small}, i.e.,\ all the $\Hom$ are sets, this can be done using simply the classical axiom of choice. If $\mathcal C$ is not locally small,
    then $\{(a,f)\}$ it's a proper class, therefore we need to use a stronger version of $\mathsf{AC}$, so some form of global choice.\newline
    Lastly, the $J_A(\cdot,\cdot)$, that we've defined, is an “external” function used to describe a WNNO/NNO, shortly we will see that we can define an “internal function”,
    in the category itself, instead of its $\Hom$-sets, which is more useful for our purposes.
\end{notation}

Let's see that if a cartesian closed category has a NNO/WNNO, then a polynomial category over it has the same NNO/WNNO.

\begin{proposition}[NNO/WNNO in polynomial categories]
    If a cartesian closed category $\mathcal{C}$ has a natural number object (resp.\ a weak natural number object), and if $x : 1 \to A$
    is an indeterminate arrow, then the cartesian closed category $\mathcal{C}[x]$ has the same natural number object (resp.\ weak natural number object).
\end{proposition}

\begin{proof}
    Let $(N,0,s)$ be a NNO (resp. WNNO) in $\mathcal{C}$, and let's consider the diagram in $\mathcal{C}[x]$:
    \[ 1 \xrightarrow{\beta(x)} B \xrightarrow{\varphi(x)} B. \]
    We're looking for a polynomial arrow $\chi(x) : N \to B$ such that the following diagram commutes in $\mathcal{C}[x]$:
    % https://q.uiver.app/#q=WzAsNixbMCwwLCIxIl0sWzEsMCwiTiJdLFsyLDAsIk4iXSxbMCwxLCIxIl0sWzEsMSwiQiJdLFsyLDEsIkIiXSxbMCwxLCIwIl0sWzEsMiwicyJdLFs0LDUsIlxcdmFycGhpKHgpIiwyXSxbMSw0LCJcXGNoaSh4KSIsMix7InN0eWxlIjp7ImJvZHkiOnsibmFtZSI6ImRhc2hlZCJ9fX1dLFsyLDUsIlxcY2hpKHgpIiwyLHsic3R5bGUiOnsiYm9keSI6eyJuYW1lIjoiZGFzaGVkIn19fV0sWzMsNCwiXFxiZXRhKHgpIiwyXSxbMCwzLCIiLDIseyJvZmZzZXQiOjEsInN0eWxlIjp7ImhlYWQiOnsibmFtZSI6Im5vbmUifX19XSxbMCwzLCIiLDEseyJvZmZzZXQiOi0xLCJzdHlsZSI6eyJoZWFkIjp7Im5hbWUiOiJub25lIn19fV1d
    % shift left and right to the arrows from 1 to 1
    \[\begin{tikzcd}
        1 & N & N \\
        1 & B & B
        \arrow["0", from=1-1, to=1-2]
        \arrow[shift right = 0.20ex, no head, from=1-1, to=2-1]
        \arrow[shift left = 0.20ex, no head, from=1-1, to=2-1]
        \arrow["s", from=1-2, to=1-3]
        \arrow["{\chi(x)}"', dashed, from=1-2, to=2-2]
        \arrow["{\chi(x)}"', dashed, from=1-3, to=2-3]
        \arrow["{\beta(x)}"', from=2-1, to=2-2]
        \arrow["{\varphi(x)}"', from=2-2, to=2-3]
    \end{tikzcd},\]
    i.e.,\ $\chi(x) \circ 0 \underset{x}{=} \beta(x)$ and $\chi(x) \circ s \underset{x}{=} \varphi(x) \circ \chi(x)$. Applying the functional completeness, \Cref{thm:functional-completeness}, to these polynomial arrows, we obtain:
    \begin{itemize}
        \item $b := \kappa_{x \in A}(\beta(x)) : A \times 1 \to B$;
        \item $f := \kappa_{x \in A}(\varphi(x)) : A \times B \to B$;
        \item $h := \kappa_{x \in A}(\chi(x)) : A \times N \to B$.
    \end{itemize}
    Now, by the uniqueness given by functional completeness, finding $\chi(x)$ is equivalent to finding $h$. We can also apply the functional completeness
    to the commutativity of the diagram above, obtaining:
    \[ h \circ \langle \pi_{A,1}, 0 \circ \pi_{A,1}' \rangle = b \qquad \text{and} \qquad h \circ \langle \pi_{A,N}, s \circ \pi_{A,N}' \rangle = f \circ \langle \pi_{A,N}, h \rangle, \]
    or, equivalently:
    \[ h \circ (1_A \times 0) = b \qquad \text{and} \qquad h \circ (1_A \times s) = f \circ \langle \pi_{A,N}, h \rangle. \]
    Let $\sigma_{1,A} := \langle \pi_{1,A}', \pi_{1,A} \rangle : 1 \times A \to A \times 1$ be the isomorphism given by the symmetry of the product. We can take the first equation, compose it with $\sigma_{1,A}$ and apply the currying isomorphism to obtain:
    \[  \Phi(h \circ (1_A \times 0) \circ \sigma_{1,A}) = \Phi(b \circ \sigma_{1,A}) =: b' : 1 \to B^A, \]
    since $\Phi$ is a bijection, $b'$ is uniquely determined by $b$. In the LHS, we have $(1_A \times 0) \circ \sigma_{1,A} = \sigma_{N,A} \circ (0 \times 1_A)$, where $\sigma_{N,A} : N \times A \to A \times N$ is the symmetry of the product.
    Moreover, by the distributivity of the currying, \Cref{prop:distributivity-of-currying}, $\Phi(h \circ \sigma_{N,A} \circ (0 \times 1_A)) = \Phi(h \circ \sigma_{N,A}) \circ 0$, therefore if we write $h' := \Phi(h \circ \sigma_{N,A}) : N \to B^A$, which is, as before, uniquely determined by $h$, we find that the first equation becomes:
    \[ h' \circ 0 = b'. \]
    Similarly, we precompose the second equation with $\sigma_{N,A} : N \times A \to A \times N$ and apply the currying; this way we get arrows $N \to B^A$ on both sides. The LHS becomes
    \[ \Phi(h \circ (1_A \times s) \circ \sigma_{N,A}) = \Phi(h \circ \sigma_{N,A} \circ (s \times 1_A)) = h' \circ s, \]
    where $(1_A \times s) \circ \sigma_{N,A} = \sigma_{N,A} \circ (s \times 1_A)$ follows from the definition of $\sigma_{N,A}$, and $h' := \Phi(h \circ \sigma_{N,A})$. The RHS becomes
    \[ \Phi(f \circ \langle \pi_{A,N}, h \rangle \circ \sigma_{N,A}), \]
    heuristically, we would like to get $f' \circ h'$, for some definition of $f' : B^A \to B^A$.
    If we find a map $q$ such that $q \circ (h' \times 1_A) = f \circ \langle \pi_{A,N}, h \rangle \circ \sigma_{N,A}$, then we can use the distributivity of the currying to get $\Phi(q) \circ h'$.\newline
    Taking $q := f \circ \langle \pi_{B^A,A}', \ev_{B,A}\rangle : B^A \times A \to B$, we have
    \begin{align*}
        q \circ (h' \times 1_A) &= f \circ \langle \pi_{B^A,A}', \ev_{B,A}\rangle \circ (h' \times 1_A) \\
                                &= f \circ \langle \pi_{N,A}',\Psi(h') \rangle \\
                                &= f \circ \langle \pi_{N,A}', \Psi(\Phi(h \circ \sigma_{N,A})) \rangle \\
                                &= f \circ \langle \pi_{N,A}', h \circ \sigma_{N,A} \rangle \\
                                &= f \circ \langle \pi_{A,N}, h \rangle \circ \sigma_{N,A},
    \end{align*}
    therefore, by taking $f' := \Phi(q)$, uniquely determined by $f$, we finally have that the second equation in the form $h' \circ s = f' \circ h'$. We now want the following equations to hold in $\mathcal{C}$:
    \[ h' \circ 0 = b' \qquad\text{and}\qquad h' \circ s = f' \circ h'. \]
    These equations corresponds to the commutativity of the diagram:
    % https://q.uiver.app/#q=WzAsNixbMCwwLCIxIl0sWzEsMCwiTiJdLFsyLDAsIk4iXSxbMSwxLCJCXkEiXSxbMiwxLCJCXkEiXSxbMCwxLCIxIl0sWzAsMSwiMCJdLFsxLDIsInMiXSxbNSwzLCJiJyIsMl0sWzAsNSwiIiwxLHsib2Zmc2V0IjoxLCJzdHlsZSI6eyJoZWFkIjp7Im5hbWUiOiJub25lIn19fV0sWzAsNSwiIiwxLHsib2Zmc2V0IjotMSwic3R5bGUiOnsiaGVhZCI6eyJuYW1lIjoibm9uZSJ9fX1dLFszLDQsImYnIiwyXSxbMSwzLCJoJyIsMix7InN0eWxlIjp7ImJvZHkiOnsibmFtZSI6ImRhc2hlZCJ9fX1dLFsyLDQsImgnIiwwLHsic3R5bGUiOnsiYm9keSI6eyJuYW1lIjoiZGFzaGVkIn19fV1d

        1 & N & N \\
        1 & {B^A} & {B^A}
        \arrow["0", from=1-1, to=1-2]
        \arrow[shift right = 0.20ex, no head, from=1-1, to=2-1]
        \arrow[shift left = 0.20ex, no head, from=1-1, to=2-1]
        \arrow["s", from=1-2, to=1-3]
        \arrow["{h'}"', dashed, from=1-2, to=2-2]
        \arrow["{h'}", dashed, from=1-3, to=2-3]
        \arrow["{b'}"', from=2-1, to=2-2]

WNNO) in $\mathcal{C}$, there exists an arrow $h' : N \to B^A$ that makes the diagram above commute, and by the functional completeness and the bijection used we can uniquely determine $\chi(x) : N \to B$ in $\mathcal{C}[x]$ such that the original diagram commutes.\newline

\end{proof}

\begin{remark}[On the choice of $q$]
    The heuristic behind the choice of $q = f \circ \langle \pi_{B^A,A}', \ev_{B,A}\rangle : B^A \times A \to B$ is pointwise. Using for a moment the set-theoretic interpretation, an element $g \in B^A$ is a map $g : A \to B$, and
    \[ q(g,a) = f(a,g(a)). \]
    Therefore, for $(n,a) \in N \times A$, we have
    \[ \left(q \circ (h' \times 1_A)\right)(n,a) = q(h'(n),a) = f(a,h'(n)(a)). \]
    Since $\Psi(h') = h \circ \sigma_{N,A}$, we have $h'(n)(a) = h(a,n)$. That is precisely the value of $f \circ \langle \pi_{A,N}, h \rangle \circ \sigma_{N,A}$ at $(n,a)$.
\end{remark}

As anticipated, the notation $J_A(a,f)$ can be replaced by a more useful “internal” notation.
Now that we have the proposition above, we can define, for each object, an arrow in the category itself, which is a sort of “function” that takes as arguments an element and a map, and returns an arrow that makes the diagram commute.

\begin{definition}[Parameterized iterator]
    Let $(N,0,s)$ be a WNNO in $\mathcal{C}$, $B$ an object of $\mathcal{C}$ and $x : 1 \to B \times B^B$ an indeterminate arrow.
    By the preceding proposition, $(N,0,s)$ is also a WNNO in $\mathcal{C}[x]$. In this category, let
    \[ y(x) := \pi_{B,B^B} \circ x : 1 \to B \qquad\text{and}\qquad v(x) := \pi_{B,B^B}' \circ x : 1 \to B^B. \]
    Since $v(x)^\sharp : B \to B$, we can choose an arrow $\chi_B(x) : N \to B$ in $\mathcal{C}[x]$ such the following diagram commutes

    \[\begin{tikzcd}
        1 & N & N \\
        1 & B & B
        \arrow["0", from=1-1, to=1-2]
        \arrow[shift right = 0.20ex, no head, from=1-1, to=2-1]
        \arrow[shift left = 0.20ex, no head, from=1-1, to=2-1]
        \arrow["s", from=1-2, to=1-3]
        \arrow["{\chi_B(x)}"', dashed, from=1-2, to=2-2]
        \arrow["{\chi_B(x)}"', dashed, from=1-3, to=2-3]
        \arrow["{y(x)}"', from=2-1, to=2-2]
        \arrow["{v(x)^\sharp}"', from=2-2, to=2-3]
    \end{tikzcd},\]
    i.e.,
    \[ \chi_B(x) \circ 0 \underset{x}{=} y(x) \qquad\text{and}\qquad
    \chi_B(x) \circ s \underset{x}{=} v(x)^\sharp \circ \chi_B(x). \]
    Functional completeness associates with this choice the unique arrow
    \[ I_B := \kappa_{x \in B \times B^B}(\chi_B(x)) : (B \times B^B) \times N \to B \]
    in $\mathcal{C}$ such that
    \[ I_B \circ \langle x \circ !_N,1_N\rangle \underset{x}{=} \chi_B(x). \]
    We call $I_B$ the \emph{parameterized iterator} for $B$.
\end{definition}

\begin{remark}
    Following the definition, if we have $p : 1 \to B \times B^B$ in $\mathcal{C}$, then $h_p := I_B \circ \langle p \circ !_N,1_N\rangle : N \to B$ is an arrow that makes the diagram commute.
    At the moment, it's as if we had a “function” $x \mapsto I_B(x,\cdot)$, which associates to each pair $(b,f)$, encoded by $x$ as $\langle a, \ulcorner f \urcorner \rangle$, an arrow $h : N \to B$ that makes the diagram commute.
    We want to get something a little bit more practical, i.e.,\ a “function” which separately takes $a$ and $f$ as arguments, and returns an arrow $h : N \to B$ that makes the diagram commute.
\end{remark}

Let's consider $y : 1 \to B$, $v : 1 \to B^B$ and $z : 1 \to N$ indeterminate arrows, and let's also define $\mathcal{C}[y,v] := \mathcal{C}[y][v]$ and $\mathcal{C}[y,v,z] := \mathcal{C}[y][v][z]$.\newline
The universal property of polynomial categories gives a canonical isomorphism $F : \mathcal{C}[x] \xrightarrow{\sim} \mathcal{C}[y,v]$ over $\mathcal{C}$, determined by $x \mapsto \langle y,v\rangle$;
its inverse sends $y$ to $y(x)$ and $v$ to $v(x)$; therefore, $F = \widehat{\iota_{y,v}}$.
By the universal property, $F$ fixes the objects and arrows of $\mathcal C$, since $\iota_{y,v}$ does so, in particular $B$, $N$, $0$, $s$ and $I_B$. Moreover, since it is a cartesian closed functor:
\[ F(y(x)) = y, \qquad F(v(x)) = v \qquad\text{and}\qquad F(v(x)^\sharp) = F(v(x))^\sharp = v^\sharp. \]
Let
\[ \widetilde{\chi}_B(y,v) := F(\chi_B(x)) : N \to B. \]
Since $F$ is a functor, the two equations satisfied by $\chi_B(x)$ give
\[ \widetilde{\chi}_B(y,v) \circ 0 \underset{y,v}{=} y \qquad\text{and}\qquad
\widetilde{\chi}_B(y,v) \circ s \underset{y,v}{=} v^\sharp \circ \widetilde{\chi}_B(y,v). \]
Thus $\widetilde{\chi}_B(y,v)$ is a solution of the corresponding recursion diagram in $\mathcal C[y,v]$. Moreover, if we apply $F$ to the defining equation for $I_B$, we get
\[ \widetilde{\chi}_B(y,v) \underset{y,v}{=} I_B \circ \langle \langle y,v\rangle \circ !_N,1_N\rangle.\]
We now have the object we were looking for, that separately takes $y$ and $v$ as arguments. If we add the indeterminate $z : 1 \to N$, precomposing with $z$ gives
\begin{align*}
    \widetilde{\chi}_B(y,v) \circ z &\underset{y,v,z}{=} I_B \circ \langle \langle y,v\rangle \circ !_N,1_N\rangle \circ z \\
                                     &\underset{y,v,z}{=} I_B \circ \langle \langle y,v\rangle,z\rangle,
\end{align*}
and using the notation $\langle y,v,z\rangle := \langle \langle y,v\rangle, z\rangle : 1 \to (B \times B^B) \times N$ gives the fundamental identity in $\mathcal{C}[y,v,z]$:
\[ \widetilde{\chi}_B(y,v) \circ z \underset{y,v,z}{=} I_B \circ \langle y,v,z\rangle. \]

Now that we have the parameterized iterator in the form we wanted, we can give an internal characterization of a WNNO in a cartesian closed category.

\begin{corollary}[Internal form of a WNNO]
    An object $N$ equipped with arrows
    \[ 1 \xrightarrow{0} N \xrightarrow{s} N \]
    is a WNNO in a cartesian closed category $\mathcal{C}$ if and only if, for every object $B$ of $\mathcal{C}$, there exists an arrow $I_B : (B \times B^B) \times N \to B$ such that, with the preceding notation for $y$, $v$ and $z$:
    \begin{align*}
        I_B \circ \langle y,v,0\rangle &\underset{y,v}{=} y, \\
        I_B \circ \langle y,v,s \circ z\rangle &\underset{y,v,z}{=} v^\sharp \circ I_B \circ \langle y,v,z\rangle.
    \end{align*}
\end{corollary}

\begin{proof}
    Let's prove the two implications separately.
    \begin{itemize}
        \item[$\boxed{\Longrightarrow}$] Let's assume that $(N,0,s)$ is a WNNO in $\mathcal{C}$. By the construction above, we have an arrow $I_B : (B \times B^B) \times N \to B$, which satisfies the identity
        \[ \widetilde{\chi}_B(y,v) \circ z \underset{y,v,z}{=} I_B \circ \langle y,v,z\rangle, \]
        in $\mathcal{C}[y,v,z]$. If we apply the substitution functor $S[0/z] : \mathcal{C}[y,v,z] \to \mathcal{C}[y,v]$ to both sides, we get
        \[ \widetilde{\chi}_B(y,v) \circ 0 \underset{y,v}{=} I_B \circ \langle y,v,0\rangle. \]
        Since $\widetilde{\chi}_B(y,v) \circ 0 \underset{y,v}{=} y$, we have
        \[ y \underset{y,v}{=} I_B \circ \langle y,v,0\rangle. \]
        Similarly, let's temporarily use $t : 1 \to N$ instead of $z$ in the identity above, and apply the substitution functor $S[s \circ z/t] : \mathcal{C}[y,v,t] \to \mathcal{C}[y,v,z]$ to both sides; we get
        \[ \widetilde{\chi}_B(y,v) \circ s \circ z \underset{y,v,z}{=} I_B \circ \langle y,v,s \circ z\rangle. \]
        Using the fact that $\widetilde{\chi}_B(y,v) \circ s \underset{y,v}{=} v^\sharp \circ \widetilde{\chi}_B(y,v)$, we have
        \[ v^\sharp \circ \widetilde{\chi}_B(y,v) \circ z \underset{y,v,z}{=} I_B \circ \langle y,v,s \circ z\rangle, \]
        and, finally, using again the initial identity, $\widetilde{\chi}_B(y,v) \circ z \underset{y,v,z}{=} I_B \circ \langle y,v,z\rangle$, we find
        \[ v^\sharp \circ I_B \circ \langle y,v,z\rangle \underset{y,v,z}{=} I_B \circ \langle y,v,s \circ z\rangle. \]
        \item[$\boxed{\Longleftarrow}$] Conversely, suppose that for every object $B$ of $\mathcal{C}$ there exists an arrow $I_B : (B \times B^B) \times N \to B$ such that the two polynomial identities in the statement are true.
        Given a diagram
        % https://q.uiver.app/#q=WzAsNixbMCwxLCIxIl0sWzEsMSwiQiJdLFsyLDEsIkIiXSxbMCwwLCIxIl0sWzEsMCwiTiJdLFsyLDAsIk4iXSxbMyw0LCIwIl0sWzQsNSwicyJdLFswLDEsImIiLDJdLFsxLDIsImYiLDJdLFszLDAsIiIsMix7Im9mZnNldCI6MSwic3R5bGUiOnsiaGVhZCI6eyJuYW1lIjoibm9uZSJ9fX1dLFszLDAsIiIsMSx7Im9mZnNldCI6LTEsInN0eWxlIjp7ImhlYWQiOnsibmFtZSI6Im5vbmUifX19XV0=
        % shift left and right to the arrows from 1 to 1
        \[\begin{tikzcd}

            1 & B & B
            \arrow["0", from=1-1, to=1-2]
            \arrow[shift right = 0.20ex, no head, from=1-1, to=2-1]
            \arrow[shift left = 0.20ex, no head, from=1-1, to=2-1]
            \arrow["s", from=1-2, to=1-3]
            \arrow["b"', from=2-1, to=2-2]

        \end{tikzcd},\]
        we need to find an arrow $h : N \to B$ that makes it commute. Let $p := \langle b,\ulcorner f\urcorner\rangle : 1 \to B \times B^B$ and define
        \[ h := I_B \circ \langle p \circ !_N,1_N\rangle = I_B \circ \langle b \circ !_N, \ulcorner f\urcorner \circ !_N, 1_N\rangle : N \to B. \]
        Precomposing and using the first polynomial identity, in $\mathcal C[b,\ulcorner f\urcorner]$, we have
        \begin{align*}
            h \circ 0  &= I_B \circ \langle p \circ !_N,1_N\rangle \circ 0 \\
                       &= I_B \circ \langle p, 0\rangle \\
                       &= I_B \circ \langle b, \ulcorner f\urcorner, 0\rangle \\
                       &= b,
        \end{align*}
        similarly, using the second one in $\mathcal C[b,\ulcorner f\urcorner,z]$, we have
        \begin{align*}
            h \circ s \circ z  &\underset{z}{=} I_B \circ \langle p, s \circ z\rangle \\
                               &\underset{z}{=} I_B \circ \langle b, \ulcorner f\urcorner, s \circ z\rangle \\
                               &\underset{z}{=} \ulcorner f\urcorner^\sharp \circ I_B \circ \langle b, \ulcorner f\urcorner, z\rangle \\
                               &\underset{z}{=} f \circ I_B \circ \langle p, z\rangle \\
                               &\underset{z}{=} f \circ I_B \circ \langle p \circ !_N,1_N\rangle \circ z = f \circ h \circ z.
        \end{align*}
        We got the pointwise commutativity of the diagram
        \[ (h \circ s) \circ z \underset{z}{=} (f \circ h) \circ z. \]
        By Lambek's normal form, \Cref{cor:lambeks-normal-form}, with respect to the indeterminate $z : 1 \to N$, we know that $\Hom_{\mathcal C}(N,B) \cong \Hom_{\mathcal C[z]}(1,B)$;
        therefore, by this isomorphism, we have $h \circ s = f \circ h$, and the diagram commutes.
    \end{itemize}
\end{proof}

Before ending this section, we give the following definitions, which will be useful to formulate the main theorem of this chapter.

\begin{definition}[Cartesian closed functor preserving a WNNO]
    Let $F : \mathcal C \to \mathcal D$ be a cartesian closed functor and let $(N,0,s)$ be a WNNO in $\mathcal C$. We say that $F$ \emph{preserves} $(N,0,s)$ if
    \[ 1_{\mathcal D} = F(1_{\mathcal C}) \xrightarrow{F(0)} F(N) \xrightarrow{F(s)} F(N) \]
    is a WNNO in $\mathcal D$.
    If $\mathcal C$ and $\mathcal D$ are equipped with chosen WNNOs $(N_{\mathcal C},0_{\mathcal C},s_{\mathcal C})$ and $(N_{\mathcal D},0_{\mathcal D},s_{\mathcal D})$, respectively, we say that $F$ \emph{strictly preserves the chosen WNNOs} if
    \[ F(N_{\mathcal C}) = N_{\mathcal D}, \qquad F(0_{\mathcal C}) = 0_{\mathcal D} \qquad\text{and}\qquad F(s_{\mathcal C}) = s_{\mathcal D}. \]
\end{definition}

\begin{definition}[Cartesian closed functor preserving an NNO]
    Let $F : \mathcal C \to \mathcal D$ be a cartesian closed functor and let $(N,0,s)$ be an NNO in $\mathcal C$. We say that $F$ \emph{preserves} $(N,0,s)$ if
    \[ 1_{\mathcal D} = F(1_{\mathcal C}) \xrightarrow{F(0)} F(N) \xrightarrow{F(s)} F(N) \]
    is an NNO in $\mathcal D$.
    If $\mathcal C$ and $\mathcal D$ are equipped with chosen NNOs $(N_{\mathcal C},0_{\mathcal C},s_{\mathcal C})$ and $(N_{\mathcal D},0_{\mathcal D},s_{\mathcal D})$, respectively, we say that $F$ \emph{strictly preserves the chosen NNOs} if
    \[ F(N_{\mathcal C}) = N_{\mathcal D}, \qquad F(0_{\mathcal C}) = 0_{\mathcal D} \qquad\text{and}\qquad F(s_{\mathcal C}) = s_{\mathcal D}. \]
\end{definition}

\begin{remark}[Images of recursion diagrams]
    Let $F : \mathcal C \to \mathcal D$ be a cartesian closed functor, and consider arrows $0 : 1_{\mathcal C} \to N$, $s : N \to N$, $a : 1_{\mathcal C} \to A$ and $f : A \to A$ in $\mathcal C$. Suppose that $h : N \to A$ satisfies
    \[ h \circ 0 = a \qquad\text{and}\qquad h \circ s = f \circ h. \]
    Functoriality gives
    \[ F(h) \circ F(0) = F(a) \qquad\text{and}\qquad F(h) \circ F(s) = F(f) \circ F(h), \]
    so $F(h)$ is a solution of the image recursion diagram. This alone does not show that $F$ preserves $(N,0,s)$: a recursion diagram in $\mathcal D$ need not be the image of one in $\mathcal C$. Moreover, when $(N,0,s)$ is only a WNNO, $F(h)$ need not coincide with a solution chosen independently in $\mathcal D$.
\end{remark}

\begin{definition}[$\mathsf{CCC}_N$]
    We denote by $\mathsf{CCC}_N$ the category whose objects are tuples
    \[ \left(\mathcal C,N_{\mathcal C},0_{\mathcal C},s_{\mathcal C},(I_A^{\mathcal C})_{A \in \ob(\mathcal C)}\right), \]
    where $\mathcal C$ is a cartesian closed category, $(N_{\mathcal C},0_{\mathcal C},s_{\mathcal C})$ is a chosen WNNO and
    \[ I_A^{\mathcal C} : (A \times A^A) \times N_{\mathcal C} \to A \]
    is a chosen parameterized iterator satisfying the identities of the preceding corollary, for every object $A$ of $\mathcal C$.
    A morphism between two such objects, based on $\mathcal C$ and $\mathcal D$, is a cartesian closed functor $F : \mathcal C \to \mathcal D$ that strictly preserves the chosen WNNOs and satisfies
    \[ F(I_A^{\mathcal C}) = I_{F(A)}^{\mathcal D} \]
    for every object $A$ of $\mathcal C$. Identities and composition are inherited from $\mathsf{CCC}$.
\end{definition}

\begin{notation}
    Sometimes we will denote an object of $\mathsf{CCC}_N$ just by the underlying cartesian closed category $\mathcal C$.
\end{notation}

\begin{remark}
    For a NNO, the parameterized iterators are unique, so their preservation follows from the preservation of the NNO. Therefore, we can just ask the morphisms to preserve the NNO, without specifying the parameterized iterator. For a WNNO, preserving the chosen parameterized iterators is an additional condition.
\end{remark}

\section{Typed \texorpdfstring{$\lambda$}{lambda}-calculi}
Now that we have seen cartesian closed categories with WNNOs, we know categories that are models of positive intuitionistic propositional
logic, and have a notion of recursion. We also want to see how this kind of categories can interpret typed $\lambda$-calculus,
and to see the latter as a sort of “internal language” of cartesian closed categories.

The specific kind of typed $\lambda$-calculus we will consider is a \emph{typed $\lambda\beta\eta$-calculus}, with \emph{terminal type}, \emph{product types}, \emph{surjective pairing} and \emph{iterator}.
This will be the first step towards establishing Lambek's equivalence of categories.

\begin{definition}
    A \emph{typed $\lambda$-calculus} is a triple $(\textsf{Ty},\textsf{Tm},\textsf{Eq})$, made of \emph{types}, \emph{terms} and \emph{equations} between terms,
    which are said to \emph{hold}, all subject to certain closure conditions. We write $a \in A$ to say that $a$ is a term of type $A$.
    The types are defined recursively as follows:
    \begin{itemize}
        \item[(Ty1)] there are $1,N$ in $\textsf{Ty}$, called \emph{basic types};
        \item[(Ty2)] if $A$ and $B$ are types, then $A \times B$ is a type;
        \item[(Ty3)] if $A$ and $B$ are types, then $B^A$ is a type.
    \end{itemize}
    The class of terms is defined recursively as follows:
    \begin{itemize}
        \item[(Tm1)] for each type $A$, there are countably many variables, $x_i^A \in A$, of type $A$;
        \item[(Tm2)] the type $1$ has a term $* \in 1$;
        \item[(Tm3a)] if $a \in A$ and $b \in B$, then there is a term $\langle a,b\rangle$ of type $A \times B$; (\emph{pairing})
        \item[(Tm3b)] if $c \in A \times B$, then there are $\pi_{A,B}(c) \in A$ and $\pi_{A,B}'(c) \in B$;
        \item[(Tm4)] if $f \in B^A$ and $a \in A$, then $\ev_{A,B}(f,a)$ is a term of type $B$; (\emph{evaluation})
        \item[(Tm5)] if $x \in A$ is a variable and $\varphi(x) \in B$ is a term, then $\lambda_{x \in A} \varphi(x) \in B^A$;\newline (\emph{$\lambda$-abstraction})
        \item[(Tm6a)] there is a term $0 \in N$;
        \item[(Tm6b)] if $n \in N$, then $s(n) \in N$;
        \item[(Tm7)] if $a \in A$, $f \in A^A$ and $n \in N$, then $I_A(a,f,n) \in A$.
    \end{itemize}
    In (Tm5) we mean that there can be a free occurrence of $x$ in $\varphi(x)$.
    We'll often abbreviate $\ev_{A,B}(f,a)$ as $f^\sharp(a)$, when the types are clear from the context.\\
    We now define recursively the class of equations between terms:
    \begin{itemize}
        \item[(Eq0)] every equation between terms is of the form $a \underset{X}{=} b$, where $a$ and $b$ are terms of the same type,
        and $X$ is a finite set of variables containing all the free variables of $a$ and $b$;
        \item[(Eq1)] $\underset{X}{=}$ is a binary relation on terms of the same type, specifically, it is an equivalence relation;
        \item[(Eq2)] if $a \underset{X}{=} b$ and $X \subseteq Y$, then $a \underset{Y}{=} b$;
        \item[(Eq3a)] if $a \underset{X}{=} b$, then $f^\sharp(a) \underset{X}{=} f^\sharp(b)$, for any $f \in B^A$ and $a,b \in A$;
        \item[(Eq3b)] if $\varphi(x) \underset{X \cup\{x\}}{=} \varphi'(x)$, then $\lambda_{x \in A} \varphi(x) \underset{X}{=} \lambda_{x \in A} \varphi'(x)$, for a variable $x \in A$ and $\varphi(x),\varphi'(x) \in B$;
        \item[(Eq4)] $a \underset{X}{=} *$, for any $a \in 1$;
        \item[(Eq5a)] $\pi_{A,B}(\langle a,b\rangle) \underset{X}{=} a$, for all $a \in A$ and $b \in B$;
        \item[(Eq5b)] $\pi_{A,B}'(\langle a,b\rangle) \underset{X}{=} b$, for all $a \in A$ and $b \in B$;
        \item[(Eq5c)] $\langle \pi_{A,B}(c),\pi_{A,B}'(c)\rangle \underset{X}{=} c$, for all $c \in A \times B$; (\emph{surjective pairing})
        \item[(Eq6a)] $(\lambda_{x \in A} \varphi(x))^\sharp(a) \underset{X}{=} \varphi(a)$, for all $a \in A$, which are substitutable for $x$ in $\varphi(x)$; (\emph{$\beta$-reduction})
        \item[(Eq6b)] $\lambda_{x \in A}(f^\sharp(x)) \underset{X}{=} f$, for all $f \in B^A$ such that $x \notin X$; (\emph{$\eta$-reduction})
        \item[(Eq7a)] $I_A(a,f,0) \underset{X}{=} a$, for all $a \in A$, $f \in A^A$ and $0 \in N$;
        \item[(Eq7b)] $I_A(a,f,s(n)) \underset{X}{=} f^\sharp(I_A(a,f,n))$, for all $a \in A$, $f \in A^A$, $n \in N$ and $s(n) \in N$;
        \item[(Eq8)] $\lambda_{x \in A} \varphi(x) \underset{X}{=} \lambda_{x' \in A} \varphi(x')$, if $x'$ is substitutable for $x$ in $\varphi(x)$ and $x'$ is not free in $\varphi(x)$; (\emph{$\alpha$-conversion}).
    \end{itemize}
\end{definition}

\begin{remark}[On the definition of typed $\lambda$-calculi]
    There may be other basic types, terms, type-forming rules, term-forming rules and equations.
    What we defined is a \emph{simply typed $\lambda$-calculus} with product types and iterator, that any other typed $\lambda$-calculus, also with product types and iterator, can be seen as an extension of.
\end{remark}

\begin{remark}
    (Eq8), the $\alpha$-conversion rule, may be omitted if we are willing to identify terms that differ only by the names of their bound variables.
\end{remark}

\begin{notation}[Free variables and closed terms]
    We denote the free variables of a term $t$, with $\FV(t)$. The definition of free variables is the standard one; see for example \cite{Carpenter1997}.
    We also say that a term $t$ is \emph{closed} if $\FV(t) = \varnothing$, and that a term $t$ is \emph{substitutable} for a variable $x$ in another term $\varphi(x)$,
    if no free occurrence of a variable in $t$ becomes bound in $\varphi(t)$.
\end{notation}

The rule (Eq2) already allows us to pass from $a \underset{X}{=} b$ to $b \underset{X \cup \{x\}}{=} a$.
Under certain assumptions, we can go the opposite way, as we see in the following propositions.

\begin{proposition}[Equality implies pointwise equality]
    In any typed $\lambda$-calculus, if $\varphi(x) \underset{X \cup \{x\}}{=} \psi(x)$, for $x \in A$, then $\varphi(a) \underset{X}{=} \psi(a)$, for all $a \in A$, assuming that:
    \begin{enumerate}[label=(\roman*)]
        \item $x \not \in X$;
        \item $\FV(a) \subseteq X$, for all $a \in A$, otherwise the thesis does not make sense.
    \end{enumerate}
\end{proposition}

\begin{proof}
    From the hypothesis, we apply the rule (Eq3b) to get
    \[ \lambda_{x \in A} \varphi(x) \underset{X}{=} \lambda_{x \in A} \psi(x). \]
    Using the evaluation at $a$, (Eq3a), and applying the $\beta$-reduction rule to both sides, we have
    \[ \varphi(a) \underset{X}{=} (\lambda_{x \in A} \varphi(x))^\sharp(a) \underset{X}{=} (\lambda_{x \in A} \psi(x))^\sharp(a) \underset{X}{=} \psi(a). \]
\end{proof}

\begin{corollary}
    If $f$ and $g$ do not contain free occurrences of $x \in A$, then $f(x) \underset{X \cup \{x\}}{=} g(x)$ implies $f \underset{X}{=} g$,
    assuming that the type is \emph{inhabited}, i.e.,\ there exists a closed term $a \in A$.
\end{corollary}

\begin{proof}
   Let $a \in A$ be a closed term. Then from $f(x) \underset{X \cup \{x\}}{=} g(x)$, using the previous proposition, we get $f(a) \underset{X}{=} g(a)$.
   Since no free occurrence of $x$ is in $f$ and $g$, we have $f(a) \underset{X}{=} f$ and $g(a) \underset{X}{=} g$ (formally by using the definition of substitution), therefore, $f \underset{X}{=} g$.
\end{proof}

It can happen that the type $A$ is \emph{empty}, or, on the other hand, that every type is inhabited, and, therefore, the previous corollary is always applicable.\newline
If we know that $f$ and $g$ are of type $B^A$, then we prove the corollary above, without the assumption that $A$ is inhabited, as follows.

\begin{proposition}
    Given $f,g \in B^A$, if $f \underset{X \cup \{x\}}{=} g$, for $x \in A$, then $f \underset{X}{=} g$.
\end{proposition}

\begin{proof}
    We'd like to use $\eta$-reduction, but we first need to go from the hypothesis to $f^\sharp(x) \underset{X \cup \{x\}}{=} g^\sharp(x)$.
    In order to do that, let's first define a new term. Let $h \in B^A$ and $x \in A$ a variable, then $h^\sharp(x) \in B$, and we can define the term:
    \[ \varphi(x) = \lambda_{h \in B^A} h^\sharp(x) \in B^{B^A}. \]
    Since $f \underset{X \cup \{x\}}{=} g$, we can apply (Eq3b) to get
    \[ (\varphi(x))^\sharp(f) \underset{X \cup \{x\}}{=} (\varphi(x))^\sharp(g), \]
    and then, by $\beta$-reduction, we have
    \[ f^\sharp(x) \underset{X \cup \{x\}}{=} g^\sharp(x). \]
    Now we can apply the $\lambda$-abstraction to both sides, since they are $A$-terms, hence
    \[ \lambda_{x \in A} (f^\sharp(x)) \underset{X}{=} \lambda_{x \in A} (g^\sharp(x)). \]
    Finally, by $\eta$-reduction, we get $f \underset{X}{=} g$.
\end{proof}

Let's see some examples of typed $\lambda$-calculi.

\begin{example}[Simply typed $\lambda$-calculus]
    As we already mentioned, if there are not types, terms, equations, type-forming rules and term-forming rules other than those in the definition above,
    we have what is called a \emph{simply typed $\lambda\beta\eta$-calculus} with terminal type, product types, surjective pairing and iterator.
    Also called \emph{pure typed $\lambda$-calculus} with \emph{weak natural number object} and all the other features above. We denote this calculus by $\mathcal L_0$.
\end{example}

\begin{example}[$\lambda$-calculus generated by a graph]
    Given a graph $\mathcal G$, the $\lambda$-calculus $\Lambda(\mathcal G)$ \emph{generated} by $\mathcal G$ is defined as it follows.
    The types and type-forming rules are the standard ones with the addition of a basic type for each vertex. The terms and term-forming rules are the standard ones
    with the addition of a new term-forming operation: if $a \in A$ is a term of type $A$ and there is an arrow $f : A \to B$ in $\mathcal G$,
    then $f(a) \in B$ is a term of type $B$. The equations are the standard ones.\newline
    We note that all vertices of $\mathcal G$ are empty types. Moreover, if we take the empty graph $\varnothing$, then $\Lambda(\varnothing) = \mathcal L_0$.
\end{example}

The next example is fundamental for this dissertation, and it's perfect as a bridge for the next section.
But before we can see it, we need to introduce the notion of morphisms between $\lambda$-calculi.

\begin{definition}[Translation between $\lambda$-calculi]
    Given two typed $\lambda$-calculi $\mathcal L$ and $\mathcal L'$, we call \emph{translation} a map $\Phi : \mathcal L \to \mathcal L'$
    such that:
    \begin{enumerate}[label=(\roman*)]
        \item $\Phi$ sends types of $\mathcal L$ to types of $\mathcal L'$ and terms of $\mathcal L$ to terms of $\mathcal L'$, such that
         if $a \in A$, then $\Phi(a) \in \Phi(A)$;
        \item $\Phi(x_i^A) = x_i^{\Phi(A)}$, for all $i \in \NN$;
    \end{enumerate}
    moreover, $\Phi$ preserves the type-forming rules and the term-forming rules:
    \begin{enumerate}[label=(\roman*),resume]
        \item $\Phi(1) = 1$ and $\Phi(N) = N$;
        \item $\Phi(A \times B) = \Phi(A) \times \Phi(B)$ and $\Phi(B^A) = \Phi(B)^{\Phi(A)}$;
        \item $\Phi(*) = *$;
        \item $\Phi(\langle a,b\rangle) = \langle \Phi(a),\Phi(b)\rangle$, furthermore, $\Phi(\pi_{A,B}(c)) = \pi_{\Phi(A),\Phi(B)}(\Phi(c))$ and $\Phi(\pi_{A,B}'(c)) = \pi_{\Phi(A),\Phi(B)}'(\Phi(c))$;
        \item $\Phi(f^\sharp(a)) = \Phi(f)^\sharp(\Phi(a))$;
        \item $\Phi(\lambda_{x \in A} \varphi(x)) = \lambda_{x \in \Phi(A)} \Phi(\varphi(x))$;
        \item $\Phi(0) = 0$ and $\Phi(s(n)) = s(\Phi(n))$;
        \item $\Phi(I_A(a,f,n)) = I_{\Phi(A)}(\Phi(a),\Phi(f),\Phi(n))$;
    \end{enumerate}
    finally, $\Phi$ preserves the equations:
    \begin{enumerate}[label=(\roman*),resume]
        \item if $a \underset{X}{=} b$, then $\Phi(a) \underset{\Phi(X)}{=} \Phi(b)$, where $\Phi(X) := \{\Phi(x) : x \in X\}$.
    \end{enumerate}
\end{definition}

There may be other things that a translation preserves, depending on the specific $\lambda$-calculi involved.

\begin{example}[Translation between graph-generated $\lambda$-calculi]
    Given a graph morphism $\varphi : \mathcal G \to \mathcal H$, we can define a translation $\Lambda(\varphi) : \Lambda(\mathcal G) \to \Lambda(\mathcal H)$, which respects
    the condition in the definition above, and sends each vertex $A$ of $\mathcal G$ to the vertex $\varphi(A)$ of $\mathcal H$, and each arrow $f : A \to B$ of $\mathcal G$ to the arrow $\varphi(f) : \varphi(A) \to \varphi(B)$ of $\mathcal H$.
    Therefore, other than such rules, it also preserves the term-forming operation $f(a) \in B$, for $a \in A$ and $f : A \to B$ in $\mathcal G$:
    \[ \Phi(f(a)) = \Phi(f)(\Phi(a)) = \varphi(f)(\Phi(a)) \in \varphi(B). \]
\end{example}

\begin{remark}[Closed terms are preserved by translations]
    It follows from the definition above that, if $a \in A$ is closed, then, since $a = a$, we have $\Phi(a) = \Phi(a)$, and therefore $\Phi(a)$ is closed too.
    Therefore, a translation sends closed terms to closed terms.
\end{remark}

\begin{remark}[Well-definedness of translations]
    Technically, we saw that $\underset{X}{=}$ is a family of equivalence relations, one for each finite set of variables $X$. Therefore, a translation $\Phi$ has to be well-defined with respect to all of them,
    and this is the reason why we require (xi) in that form in the definition above.
\end{remark}

\begin{definition}[Equality of translations]
    Given two translations $\Phi, \Psi : \mathcal L \to \mathcal L'$, we say that they are \emph{equal} if, for every type $A$ of $\mathcal L$, $\Phi(A) = \Psi(A)$, and for every term $a \in A$, $\Phi(a) \underset{\Phi(X)}{=} \Psi(a)$, where $X$ is a finite set of variables containing all the free variables of $a$.
\end{definition}

\begin{definition}[$\lambda\text{-}\mathsf{Calc}$]
    We define by $\lambda\text{-}\mathsf{Calc}$ the category of \emph{typed $\lambda$-calculi}, whose objects are typed $\lambda$-calculi that respect the definition above, possibly with additional basic types, terms, type-forming rules, term-forming rules and equations, and whose morphisms are translations between them.
    Thus, a morphism is only required to satisfy conditions (i)--(xi) above, even when its source and target have additional structure.
\end{definition}

\begin{example}[Internal language of a CCC]
    We define a functor
    \[ \mathbf L : \mathsf{CCC}_N \to \lambda\text{-}\mathsf{Calc}, \]
    called \emph{internal language} of a CCC. Given an object $\mathcal C$ of $\mathsf{CCC}_N$, the $\lambda$-calculus $\mathbf L(\mathcal C)$ is defined as follows:
    \begin{itemize}
        \item the types are the objects of $\mathcal C$, with the basic types $1$ and $N$, and the type-forming rules are the standard ones; they are defined by the cartesian closed structure of $\mathcal C$;

\section{Lambek's equivalence}
We want to show that the functor $\mathbf L : \mathsf{CCC}_N \to \lambda\text{-}\mathsf{Calc}$ is an equivalence of categories, and to do that we need to define a functor in the opposite direction.

\begin{definition}[CCC defined by a $\lambda$-calculus]
    Given a typed $\lambda$-calculus $\mathcal L$, we define a cartesian closed category $\mathbf C(\mathcal L)$ as follows:
    \begin{itemize}
        \item the objects of such category are the types of $\mathcal L$;
        \item given two objects $A$ and $B$ of this category, the arrows from $A$ to $B$ are the equivalence classes of pairs $(x, \varphi(x))$, where $x \in A$ and  $\varphi(x) \in B$ is a term with no free variables other than $x$.
        Equivalently, $\Hom_{\mathbf C(\mathcal L)}(A,B) = \{[(x,\varphi(x))]_\sim : x \in A, \varphi(x) \in B\}$, where $[(x,\varphi(x))]_\sim$ is the equivalence class of $(x,\varphi(x))$ with respect to the relation
        \[ (x,\varphi(x)) \sim (y,\psi(y)) \quad\overset{\textnormal{def}}{\iff}\quad \varphi(z) \underset{\{z\}}{=} \psi(z), \]
        for $x,y \in A$, $\varphi(x),\psi(y) \in B$ and $z \in A$ (basically, up to $\alpha$-conversion, they are the same term). The categorical structure of $\mathbf C(\mathcal L)$ can be retrieved from the type-forming rules, as follows:
        \begin{itemize}[label=$*$]
            \item $(x,x)$ is the identity arrow of $A$;
            \item given $[(x,\varphi(x))] : A \to B$ and $[(y,\psi(y))] : B \to C$, the composition is defined as
            \[ [(y,\psi(y))] \circ [(x,\varphi(x))]  := [(x,\psi(\varphi(x)))], \]
            where $\psi(\varphi(x))$ is the term obtained by substituting $\varphi(x)$ for $y$ in $\psi(y)$;
            \item by the definition of composition, it is easy to verify that the identity and associativity laws hold.
        \end{itemize}
        The cartesian closed structure is also defined by the type-forming rules, as follows:
        \begin{itemize}[label=$*$]
            \item $!_A = [(x,*)] : A \to 1$ is the unique arrow from $A$ to $1$;
            \item $\pi_{A,B} = [(z \in A \times B,\pi_{A,B}(z))] : A \times B \to A$ and $\pi_{A,B}' = [(z \in A \times B,\pi_{A,B}'(z))] : A \times B \to B$ are the projections;
            \item $\ev_{B,A} = [(y \in B^A \times A,\ev_{C,A}(\pi_{B^A,A}(y),\pi_{B^A,A}'(y)))] : B^A \times A \to B$;
            \item $\langle [(x \in C,\varphi(x) \in A)],[(y \in C,\psi(y) \in B)] \rangle = [(z,\langle \varphi(z),\psi(z) \rangle)] : C \to A \times B$ is the pairing of two arrows $[(x,\varphi(x))]$ and $[(y,\psi(y))]$;
            \item $[(z \in A \times B,\chi(z) \in C)]^* = [(x,\lambda_{y \in B} \chi(\langle x,y\rangle))] : A \to C^B$;
        \end{itemize}
        where the operations used to define the cartesian closed structure are the term-forming rules of the $\lambda$-calculus $\mathcal L$.
        Finally, the chosen WNNO and iterator are defined as:
        \begin{itemize}[label=$*$]
            \item $0 = [(*,x)] : 1 \to N$;
            \item $s = [(x,s(x))] : N \to N$;
            \item $I_A = [(w,I_B(\pi_{B,B^B}(\pi_{B \times B^B,N}(w)), \pi_{B,B^B}'(\pi_{B \times B^B,N}(w)), \pi_{B \times B^B,N}(w)))] : (B \times B^B) \times N \to B$, for $w \in (B \times B^B) \times N$.
        \end{itemize}
    \end{itemize}
    Given a translation $\Phi : \mathcal L \to \mathcal L'$, we define a cartesian closed functor that strictly preserves the chosen WNNOs and iterators, $\mathbf C(\Phi) : \mathbf C(\mathcal L) \to \mathbf C(\mathcal L')$, as follows:
    \[ \mathbf C(\Phi) := \begin{cases}
        A \mapsto \Phi(A) & \text{on the objects} \\
        [(x,\varphi(x))] \mapsto [(\Phi(x),\Phi(\varphi(x)))] & \text{on the arrows}
    \end{cases}, \]
    and easily follows from the definition of translation that $\mathbf C(\Phi)$ is well-defined and preserves the cartesian closed structure, the chosen WNNOs and iterators.
\end{definition}

Before proving the equivalence, we need to introduce one last idea: we want to find the correspondent construction, in terms of $\lambda$-calculi, of the polynomial category of a CCC.\newline
Instead of adding an indeterminate arrow $x : 1 \to A$ to the cartesian closed category $\mathbf C(\mathcal L)$, we could add a \emph{parameter} $x$ of type $A$ to the $\lambda$-calculus $\mathcal L$.

\begin{definition}[$\lambda$-calculus with a parameter]
    Given a typed $\lambda$-calculus $\mathcal L$ and a type $A$ of $\mathcal L$, we define the $\lambda$-calculus with the \emph{adjoint parameter} $x$ of type $A$, denoted by $\mathcal L(x)$,
    as the $\lambda$-calculus obtained by considering $x$ as a closed term of type $A$. Therefore, $\mathcal L(x)$ has the same types of $\mathcal L$, however, the closed terms of $\mathcal L(x)$
are the terms $\varphi(x)$ of $\mathcal L$, which contain no free variables other than $x$.
\end{definition}

It won't surprise us that $\mathbf C(\mathcal L)[x]$ and $\mathbf C(\mathcal L(x))$ are isomorphic categories.

\begin{proposition}
    $\mathbf C(\mathcal L)[x] \cong \mathbf C(\mathcal L(x))$.
\end{proposition}

\begin{definition}[Equivalence of categories]
    Given two categories $\mathcal C$ and $\mathcal D$, we say that they are \emph{equivalent}, $\mathcal C \simeq \mathcal D$, if there exist two functors $F : \mathcal C \to \mathcal D$ and $G : \mathcal D \to \mathcal C$,

\end{definition}

\begin{remark}[{\cite[Lemma~9.2.4]{nLabFunctorCategory2026}}]

\end{remark}

        Therefore, the diagram of naturality commutes, and $\varepsilon$ is a natural transformation.

        By Lambek's normal form, we know $\Hom_{\mathcal C[y]}(1,C) \cong \Hom_\mathcal C(B,C)$,
        but $\Hom_{\mathbf C \circ \mathbf L(\mathcal C)}(B,C) \cong \Hom_{\mathcal C[y]}(1,C)$, by definition, since we take the equivalence classes of terms $\varphi(y) \in C$, that are said to be equal if they are equal in $\mathcal C[y]$.
        Therefore, $\varepsilon_\mathcal C$ is an isomorphism in $\mathsf{CCC}_N$, hence $\varepsilon$ is a natural isomorphism.
        \item[$\boxed{1_{\lambda\text{-}\mathsf{Calc}} \Rightarrow \mathbf L \circ \mathbf C}$] We want to define $\eta : 1_{\lambda\text{-}\mathsf{Calc}} \Rightarrow \mathbf L \circ \mathbf C$, for each $\mathcal L \in \ob(\lambda\text{-}\mathsf{Calc})$, i.e.,
        we want a translation $\eta_\mathcal L : \mathcal L \to \mathbf L \circ \mathbf C(\mathcal L)$, and such that, given a translation $\Phi : \mathcal L \to \mathcal L'$, the diagram of naturality commutes and $\eta_{\mathcal L}$ is an isomorphism.\newline
        Let's define $\eta_\mathcal L$ as follows:
        \begin{itemize}
            \item on the types, $\eta_\mathcal L(A) := A$, for all $A \in \mathsf{Ty}(\mathcal L)$;
            \item on the terms, given a term $\varphi(X) = \varphi(x_1,\dots,x_n) \in A$, we can see it as a closed term in $\mathcal L(X)$;
            since $\mathbf C(\mathcal L(X)) \cong \mathbf C(\mathcal L)[X]$, by [REFERENCE], we get a polynomial arrow $1 \to A$, and applying $\mathbf C$ we find the arrow $[(* \in 1, \varphi(X))] : 1 \to A$ in $\mathbf C(\mathcal L)[X]$.
            Now applying $\mathbf L$, a polynomial arrow becomes a term, so find the term $[(* \in 1, \varphi(X))] \in A$ and therefore define
            \[ \eta_\mathcal L(\varphi(X)) := [(* \in 1, \varphi(X))] \in A, \]
            i.e., we get again a term of type $A$ with free variables $X = x_1,\dots,x_n$. It's easy to verify that $\eta_\mathcal L$ is a translation and that is well-defined (the equality of terms in $\mathbf L$ is defined as the equality of polynomial arrows, which is the equality defined by $\mathbf C$, etc.).
        \end{itemize}
        To check that $\eta_\mathcal L$ is an isomorphism, we construct its inverse, $\eta_\mathcal L^{-1} : \mathbf L \circ \mathbf C(\mathcal L) \to \mathcal L$, that maps $[(* \in 1, \varphi(X))]$ to $\varphi(X)$. It's easy to verify that this map its the inverse of $\eta_\mathcal L$ and it is a well-defined translation.\newline
        Finally we need to check that, given a translation $\Phi : \mathcal L \to \mathcal L'$, the diagram of naturality

        \[\begin{tikzcd}
            {\mathcal L} & {\mathbf L \circ \mathbf C(\mathcal L)} \\
            {\mathcal L'} & {\mathbf L \circ \mathbf C(\mathcal L')}
            \arrow["{\eta_{\mathcal L}}", dashed, from=1-1, to=1-2]
            \arrow["\Phi"', from=1-1, to=2-1]
            \arrow["{\mathbf L \circ \mathbf C(\Phi)}", from=1-2, to=2-2]
            \arrow["{\eta_{\mathcal L'}}"', dashed, from=2-1, to=2-2]
        \end{tikzcd}\]
        As in the previous case, on the types both composites send $A$ to $\Phi(A)$. On the terms, given $\varphi(X) \in A$, we have
        \[ \eta_{\mathcal L'} \circ \Phi(\varphi(X)) = [(* \in 1, \Phi(\varphi(X)))], \]
        while
        \[ (\mathbf L \circ \mathbf C(\Phi)) \circ \eta_\mathcal L(\varphi(X)) = (\mathbf L \circ \mathbf C(\Phi))([(* \in 1, \varphi(X))]), \]

\end{proof}

We can now give a precise meaning to the idea that a $\lambda$-calculus is an internal language of a cartesian closed category.

\begin{definition}[Interpretation of a $\lambda$-calculus in a CCC]
    We call an \emph{interpretation} of a typed $\lambda$-calculus $\mathcal L$ in a cartesian closed category $\mathcal C$ with a chosen WNNO and an iterator, either a translation $\mathcal L \to \mathbf L(\mathcal C)$
    or a cartesian closed functor that strictly preserves the chosen WNNOs and iterators $\mathbf C(\mathcal L) \to \mathcal C$.
\end{definition}

\begin{remark}[Internal language of a CCC]
    It's easy to see that $\mathcal L_0$ is an initial object in $\lambda\text{-}\mathsf{Calc}$, since for each typed $\lambda$-calculus $\mathcal L$, there exists a unique translation $\mathcal L_0 \to \mathcal L$.
    Therefore, the simple typed $\lambda$-calculus is interpreted in every cartesian closed category with a chosen WNNO and an iterator, hence it is the internal language of every CCC.
\end{remark}

The above equivalence of categories is the most general in the context of cartesian categories, we can specialize it to the case of cartesian closed categories without a chosen WNNO and iterator.
Let's first define the following.

\begin{definition}[$\lambda\text{-}\mathsf{Calc}_0$]
    We denote by $\lambda\text{-}\mathsf{Calc}_0$ the category whose objects are typed $\lambda$-calculi as defined above, but without:
    \begin{itemize}
        \item the basic type $N$ in (Ty1);
        \item the term-forming rules for zero, successor and the iterator, namely (Tm6a), (Tm6b) and (Tm7);
        \item the iterator equations (Eq7a) and (Eq7b).
    \end{itemize}
    Its morphisms are translations, as defined above, without the requirements involving $N$, $0$, $s$ and $I_A$.
\end{definition}

Using the same functors $\mathbf C$ and $\mathbf L$ defined above, with the same definitions and the same proofs, forgetting
the chosen WNNO and iterator, we have a proof of the following equivalence of categories.

\begin{theorem}[Curry--Howard--Lambek without WNNO and iterator]\label{thm:curry-howard-lambek-without-wnno-and-iterator}
    \leavevmode\newline
    $\mathsf{CCC} \simeq \lambda\text{-}\mathsf{Calc}_0$.
\end{theorem}
